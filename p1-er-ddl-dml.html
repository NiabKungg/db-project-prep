🌱 อ่าน CREATE TABLE ให้เป็นก่อน (แปลทีละบรรทัด — 5 นาทีจบ)

โค้ด CREATE TABLE ทุกก้อนในเว็บนี้ใช้โครงเดียวกันหมด — ถ้าอ่านตารางที่เล็กที่สุด (DEPARTMENTS) ออก แปลว่าอ่านอีก 10 ตารางออกทั้งหมด:

บรรทัดโค้ดแปลเป็นไทยล้วน
CREATE TABLE DEPARTMENTS ("ขอสร้างตารางใหม่ชื่อ DEPARTMENTS นิยามอยู่ในวงเล็บนี้"
DEPT_ID NUMBER(4) CONSTRAINT PK_DEPT PRIMARY KEY,"คอลัมน์แรกชื่อ DEPT_ID เก็บตัวเลขไม่เกิน 4 หลัก · ตั้งชื่อ constraint ว่า PK_DEPT · PRIMARY KEY = รหัสประจำตัวของแถว → ห้ามซ้ำ ห้ามว่าง (เหมือนเลขบัตรประชาชน)"
DEPT_NAME VARCHAR2(60) CONSTRAINT NN_DEPT_NAME NOT NULL"คอลัมน์ที่สองชื่อ DEPT_NAME เก็บข้อความยาวไม่เกิน 60 ตัวอักษร · NOT NULL = ห้ามเว้นว่าง (แผนกต้องมีชื่อเสมอ)"
);"จบนิยามตาราง — ฐานข้อมูลจะสร้างให้ทันที จากนี้พิมพ์ INSERT เพิ่มข้อมูลได้เลย"

CONSTRAINT อ่านว่า "คอนสเตรนต์" = "กฎที่ฝังไว้ในตาราง" — ให้ฐานข้อมูลเป็นคนเช็คข้อมูลเสมอ ไม่ต้องเขียนโค้ดตรวจเอง (คนพิมพ์ผิดฐานข้อมูลปฏิเสธทันที) · ชื่อ constraint ที่เราตั้ง (PK_/NN_/UQ_/CK_/FK_) จะโผล่ในผล 05_VERIFY ตอน demo จึงตั้งให้อ่านรู้เรื่อง

ชนิดข้อมูล (Type) ที่พบในงานนี้ — 5 แบบจบ

ชนิดเก็บอะไรตัวอย่างค่าจริงจากข้อมูลของเรา
NUMBER(4) / NUMBER(6)ตัวเลขจำนวนเต็ม ไม่เกิน n หลักรหัสแผนก 10 · รหัสพนักงาน 1001
NUMBER(6,2)ตัวเลขมีทศนิยม (ทั้งหมด 6 หลัก ทศนิยม 2 หลัก)ค่าโดยสาร 13.00 (Part II)
VARCHAR2(60)ข้อความยาวไม่เกิน n ตัวอักษร (ใช้บ่อยสุด)'ฝ่ายบริหาร', 'บข 1234', 'pakorn@corp.co.th'
CHAR(1)ตัวอักษร 1 ตัวตายตัว (เก็บค่า Y/N)สิทธิ์เข้าถึง 'Y'
DATEวันที่ + เวลา (DEFAULT SYSDATE = ใช้วันเวลาปัจจุบันถ้าไม่กรอก)วันที่ส่งเอกสาร, เวลากดกริ่ง
CLOBข้อความยาวมาก ๆ (เนื้อความประกาศ)'บริษัทงดให้บริการ 3 วันในช่วงสงกรานต์...'

🧹 ขั้นที่ 0 — DROP ทุกอย่างก่อน (ไฟล์ 00)

อาจารย์สั่งชัดว่า "DROP ทุกตารางแล้วสร้างใหม่ให้ดู" — ไฟล์ 00 คือขั้นนี้ จำหลักการ 3 ข้อ:

  1. ลบตารางลูกก่อนตารางแม่ — เพราะลูกมี FK อ้างถึงแม่ ถ้าลบแม่ก่อนจะโดน ORA-02292: integrity constraint violated
  2. CASCADE CONSTRAINTS PURGE — CASCADE CONSTRAINTS = ลบ FK ที่ชี้มาที่ตารางนี้ทิ้งไปด้วย (กันติดขัดจากของค้าง) · PURGE = ไม่เก็บไว้ recycle bin จะ CREATE ชื่อซ้ำได้เลย
  3. อย่าลืม DROP SEQUENCE / PROCEDURE / TRIGGER เก่าด้วย มิฉะนั้นตอนรัน 01–03 จะชนชื่อ
00_DROP_ALL.sql 
-- =====================================================================
-- 00_DROP_ALL.sql — ใช้ "DROP ทุกตารางแล้วสร้างใหม่" ตอนสาธิตหน้าห้อง

-- ลบตาราง (ลูกก่อนแม่ เพราะมี FK ผูกกัน)
DROP TABLE READ_LOGS      CASCADE CONSTRAINTS PURGE;
DROP TABLE NOTIFICATIONS  CASCADE CONSTRAINTS PURGE;
DROP TABLE DISTRIBUTIONS  CASCADE CONSTRAINTS PURGE;
DROP TABLE SIGNATURES     CASCADE CONSTRAINTS PURGE;
DROP TABLE APPROVALS      CASCADE CONSTRAINTS PURGE;
DROP TABLE DOC_STATUS_LOG CASCADE CONSTRAINTS PURGE;
DROP TABLE ACCESS_MATRIX  CASCADE CONSTRAINTS PURGE;
DROP TABLE DOCUMENTS      CASCADE CONSTRAINTS PURGE;
DROP TABLE EMPLOYEES      CASCADE CONSTRAINTS PURGE;
DROP TABLE ROLES          CASCADE CONSTRAINTS PURGE;
DROP TABLE DEPARTMENTS    CASCADE CONSTRAINTS PURGE;

-- ลบ Sequence
DROP SEQUENCE SEQ_DOC;
DROP SEQUENCE SEQ_APPROVAL;
DROP SEQUENCE SEQ_SIGN;
DROP SEQUENCE SEQ_DIST;
DROP SEQUENCE SEQ_NOTIF;
DROP SEQUENCE SEQ_LOG;

-- ลบ Procedure (ถ้ามี)
DROP PROCEDURE SP_SUBMIT_DOC;
DROP PROCEDURE SP_DECIDE_APPROVAL;
DROP PROCEDURE SP_ACKNOWLEDGE_DOC;
DROP PROCEDURE SP_CHECK_SLA_REMIND;

-- ลบ Trigger (ถ้ามี)
DROP TRIGGER TRG_DOC_STATUS_LOG;

⚠️ เรียง "ลูก (READ_LOGS) → แม่ (DEPARTMENTS)" — และถ้ารันตอนฐานข้อมูลยังไม่มีอะไรเลย บางขั้นจะ error "does not exist" ซึ่งไม่ใช่ปัญหา (DROP ของที่ไม่มี = ข้ามได้) แต่ demo ครั้งแรกควรสร้างก่อน 1 รอบ แล้วค่อยโชว์ DROP

🏗️ ขั้นที่ 1 — CREATE ตารางทีละใบ (ไฟล์ 01 · DDL)

อ่านแบบ "ตารางละ 1 ก้อน": ดูชื่อคอลัมน์ + ชนิดข้อมูล แล้วดู Constraint ที่ประกาศด้วยคำสั่ง CONSTRAINT ชื่อ ชนิด — การตั้งชื่อ constraint แบบมีระบบ (NN_ / UQ_ / CK_ / PK_ / FK_) ช่วยให้อ่านผล 05_VERIFY ได้ง่ายมาก

1. DEPARTMENTS — แผนก/หน่วยงาน

-- ---------- 1. DEPARTMENTS : แผนก/หน่วยงาน ----------
CREATE TABLE DEPARTMENTS (
    DEPT_ID      NUMBER(4)      CONSTRAINT PK_DEPT PRIMARY KEY,
    DEPT_NAME    VARCHAR2(60)   CONSTRAINT NN_DEPT_NAME NOT NULL
);
  • PK_DEPT = Primary Key — รหัสแผนกไม่ซ้ำ ไม่เป็น NULL (จะเห็นประเภท P ใน 05_VERIFY)
  • NN_DEPT_NAME = NOT NULL — แผนกต้องมีชื่อเสมอ เพราะถ้าไม่มีชื่อ การจัดกลุ่มพนักงานจะไม่มีความหมาย (ประเภท C)

2. ROLES — บทบาท 4 แบบ (แผนที่กับ Actor ใน UC Diagram)

-- ---------- 2. ROLES : บทบาทตาม Use Case (IT / Creator / Approver / Employee) ----------
CREATE TABLE ROLES (
    ROLE_ID      NUMBER(4)      CONSTRAINT PK_ROLE PRIMARY KEY,
    ROLE_NAME    VARCHAR2(40)   CONSTRAINT NN_ROLE_NAME NOT NULL CONSTRAINT UQ_ROLE_NAME UNIQUE,
    ROLE_DESC    VARCHAR2(120)
);
  • UQ_ROLE_NAME = UNIQUE — ชื่อบทบาทห้ามซ้ำ เพราะ Procedure เช่น SP_DECIDE_APPROVAL จะหา "ผู้มี ROLE_ID = 3 (APPROVER)" — ถ้าบทบาทซ้ำ สิทธิ์จะพัง

3. EMPLOYEES — พนักงาน (Actor ทุกตัว)

-- ---------- 3. EMPLOYEES : พนักงาน (Actor ทุกตัวใน UC Diagram) ----------
CREATE TABLE EMPLOYEES (
    EMP_ID       NUMBER(6)      CONSTRAINT PK_EMP PRIMARY KEY,
    EMP_NAME     VARCHAR2(80)   CONSTRAINT NN_EMP_NAME NOT NULL,
    EMAIL        VARCHAR2(80)   CONSTRAINT NN_EMP_EMAIL NOT NULL CONSTRAINT UQ_EMP_EMAIL UNIQUE,
    POSITION     VARCHAR2(60),
    DEPT_ID      NUMBER(4)      CONSTRAINT FK_EMP_DEPT REFERENCES DEPARTMENTS(DEPT_ID),
    ROLE_ID      NUMBER(4)      CONSTRAINT FK_EMP_ROLE REFERENCES ROLES(ROLE_ID),
    HIRE_DATE    DATE           DEFAULT SYSDATE
);
  • UQ_EMP_EMAIL — อีเมลไม่ซ้ำ เพราะ UC09 ใช้อีเมลเป็นชื่อบัญชีเข้าสู่ระบบ + UC01 ส่งแจ้งเตือนทางอีเมล ถ้าซ้ำจะส่งผิดคน
  • FK_EMP_DEPT / FK_EMP_ROLE = Foreign Key (ประเภท R) — พนักงานต้องอยู่แผนกและบทบาทที่ "มีจริง" ในระบบ กันพิมพ์รหัสแผนกเลขมั่ว
  • DEFAULT SYSDATE — ถ้าไม่กรอกวันเข้าทำงาน ใช้วันที่ปัจจุบันอัตโนมัติ

4. DOCUMENTS — แกนหลักของระบบ ⭐

-- ---------- 4. DOCUMENTS : ประกาศ/ข้อความเวียน (แกนหลักของระบบ) ----------
CREATE TABLE DOCUMENTS (
    DOC_ID       NUMBER(8)      CONSTRAINT PK_DOC PRIMARY KEY,
    DOC_TYPE     VARCHAR2(15)   CONSTRAINT NN_DOC_TYPE NOT NULL
                 CONSTRAINT CK_DOC_TYPE CHECK (DOC_TYPE IN ('ANNOUNCEMENT','MEMO')),
    TITLE        VARCHAR2(150)  CONSTRAINT NN_DOC_TITLE NOT NULL,
    CONTENT      CLOB,
    PRIORITY     VARCHAR2(10)   DEFAULT 'GENERAL'
                 CONSTRAINT CK_DOC_PRI CHECK (PRIORITY IN ('EMERGENCY','GENERAL')),
    STATUS       VARCHAR2(12)   DEFAULT 'DRAFT'
                 CONSTRAINT CK_DOC_STATUS CHECK (STATUS IN
                   ('DRAFT','PENDING','APPROVED','REJECTED','SIGNED','DISTRIBUTED','COMPLETED','ARCHIVED')),
    CREATED_BY   NUMBER(6)      CONSTRAINT FK_DOC_CREATOR REFERENCES EMPLOYEES(EMP_ID),
    CREATED_DATE DATE           DEFAULT SYSDATE,
    SLA_HOURS    NUMBER(3)      DEFAULT 24 CONSTRAINT CK_DOC_SLA CHECK (SLA_HOURS > 0),
    EXPIRY_DATE  DATE
);
  • CK_DOC_TYPE — เอกสารมีแค่ 2 ประเภท: ANNOUNCEMENT (ประกาศ) / MEMO (ข้อความเวียน) ตรงกับ 2 Pool ใน BPMN ของงาน
  • CK_DOC_STATUS — สถานะ 8 ค่าคือชีวิตของเอกสาร (ดูแผนภาพเต็มในหน้า PL/SQL): DRAFT → PENDING → APPROVED → SIGNED → DISTRIBUTED → COMPLETED (+ แขนง REJECTED, ARCHIVED) — ถ้าโค้ดพยายามอัปเดตสถานะออกนอกชุดนี้ DB ปฏิเสธทันที
  • CK_DOC_PRI + DEFAULT 'GENERAL' — ความเร่งด่วน 2 ค่า: EMERGENCY กระจาย 3 ช่องทาง, GENERAL กระจาย 2 ช่องทาง (SP_DECIDE_APPROVAL อ่านค่านี้)
  • SLA_HOURS DEFAULT 24 CHECK (SLA_HOURS > 0) — กรอบเวลาการอ่าน (UC04 ตั้งค่า) ต้องเป็นบวกเท่านั้น และมาจากกฎ BR-Memo-101 "ยืนยันภายใน 24 ชม."
  • CONTENT CLOB — CLOB = ข้อความยาวขนาดใหญ่ เหมาะกับเนื้อหาประกาศ

5. DOC_STATUS_LOG — Audit Trail (โต๊ะบันทึกของ Trigger)

-- ---------- 5. DOC_STATUS_LOG : ประวัติการเปลี่ยนสถานะ (Audit Trail — เกิดจาก Trigger) ----------
CREATE TABLE DOC_STATUS_LOG (
    LOG_ID       NUMBER(10)     CONSTRAINT PK_STLOG PRIMARY KEY,
    DOC_ID       NUMBER(8)      CONSTRAINT FK_STLOG_DOC REFERENCES DOCUMENTS(DOC_ID),
    OLD_STATUS   VARCHAR2(12),
    NEW_STATUS   VARCHAR2(12)   CONSTRAINT NN_STLOG_NEW NOT NULL,
    CHANGED_DATE DATE           DEFAULT SYSDATE
);
  • ทุกครั้งที่สถานะของเอกสารเปลี่ยน Trigger จะ INSERT แถวนี้อัตโนมัติ — ตอบโจทย์ UC03 "ดูสถานะย้อนหลัง" และกฎ BR-Memo-103 "ห้ามแก้ Log"

6. APPROVALS — ผลพิจารณาอนุมัติ (XOR Gateway)

-- ---------- 6. APPROVALS : ผลการพิจารณาอนุมัติ (XOR: ผ่าน/ตีกลับ) ----------
CREATE TABLE APPROVALS (
    APPROVAL_ID  NUMBER(8)      CONSTRAINT PK_APRV PRIMARY KEY,
    DOC_ID       NUMBER(8)      CONSTRAINT NN_APRV_DOC NOT NULL
                 CONSTRAINT FK_APRV_DOC REFERENCES DOCUMENTS(DOC_ID),
    APPROVER_ID  NUMBER(6)      CONSTRAINT FK_APRV_EMP REFERENCES EMPLOYEES(EMP_ID),
    DECISION     VARCHAR2(10)   CONSTRAINT NN_APRV_DEC NOT NULL
                 CONSTRAINT CK_APRV_DEC CHECK (DECISION IN ('APPROVED','REJECTED')),
    REASON       VARCHAR2(200),
    DECIDED_DATE DATE           DEFAULT SYSDATE
);
  • ตรงกับ XOR Gateway ใน BPMN: เดินได้แค่ 1 ทางเท่านั้น — APPROVED (ลงนาม+กระจาย) หรือ REJECTED (ส่งกลับให้ธุรการแก้ ตาม DR-Ann-101)
  • REASON — UC07 Extension 4a ให้ "เขียนเหตุผลการตีกลับ" คอลัมน์นี้คือที่เก็บเหตุผลนั้น (UC06 อ่านกลับไปแสดงให้ธุรการ)

7. SIGNATURES — ลายมือชื่ออิเล็กทรอนิกส์

-- ---------- 7. SIGNATURES : ลายมือชื่ออิเล็กทรอนิกส์ (ห้ามเผยแพร่ก่อนลงนาม = BR-Ann-102) ----------
CREATE TABLE SIGNATURES (
    SIGN_ID      NUMBER(8)      CONSTRAINT PK_SIGN PRIMARY KEY,
    DOC_ID       NUMBER(8)      CONSTRAINT NN_SIGN_DOC NOT NULL
                 CONSTRAINT FK_SIGN_DOC REFERENCES DOCUMENTS(DOC_ID)
                 CONSTRAINT UQ_SIGN_DOC UNIQUE,          -- ลงนามได้ 1 ครั้งต่อเอกสาร
    SIGNER_ID    NUMBER(6)      CONSTRAINT FK_SIGN_EMP REFERENCES EMPLOYEES(EMP_ID),
    SIGNATURE_HASH VARCHAR2(64) CONSTRAINT NN_SIGN_HASH NOT NULL,
    SIGNED_DATE  DATE           DEFAULT SYSDATE
);
  • UQ_SIGN_DOC — เอกสารเดียวลงนามได้แค่ 1 แถว (UNIQUE บน DOC_ID) พิสูจน์กฎ BR-Ann-102 ด้วยโครงสร้าง DB: มีลายเซ็นแล้ว = เอกสารจึงเผยแพร่ได้

-- ---------- 8. DISTRIBUTIONS : การกระจายเอกสารออกช่องทางต่าง ๆ (AND Gateway) ----------
CREATE TABLE DISTRIBUTIONS (
    DIST_ID      NUMBER(8)      CONSTRAINT PK_DIST PRIMARY KEY,
    DOC_ID       NUMBER(8)      CONSTRAINT NN_DIST_DOC NOT NULL
                 CONSTRAINT FK_DIST_DOC REFERENCES DOCUMENTS(DOC_ID),
    CHANNEL      VARCHAR2(6)    CONSTRAINT NN_DIST_CH NOT NULL
                 CONSTRAINT CK_DIST_CH CHECK (CHANNEL IN ('EMAIL','APP','SMS')),
    RECIPIENT_GROUP VARCHAR2(60) CONSTRAINT NN_DIST_GRP NOT NULL,
    SENT_DATE    DATE           DEFAULT SYSDATE,
    STATUS       VARCHAR2(6)    DEFAULT 'SENT'

);

🌱 SELECT แรกในชีวิต + อ่าน INSERT ให้เป็น (ก่อนไปหน้า PL/SQL)

ใส่ข้อมูล (INSERT) แล้วต้องดูข้อมูล (SELECT) เป็น — สามตัวอย่างนี้ใช้ข้อมูลจริงของไฟล์ 02 ทั้งหมด ผลลัพธ์บนจอจะตรงเป๊ะแบบนี้:

ตัวอย่าง 1 — ดูข้อมูลทั้งหมด

SELECT * FROM EMPLOYEES;
  • SELECT * = ขอ ทุกคอลัมน์ · FROM EMPLOYEES = จากตารางพนักงาน — ได้ 8 แถว (พนักงาน 8 คน) โผล่บนจอทั้งคน

ตัวอย่าง 2 — เลือกเฉพาะคอลัมน์ + คัดเฉพาะแถว (WHERE)

SELECT EMP_ID, EMP_NAME, EMAIL
FROM   EMPLOYEES
WHERE  ROLE_ID = 4
ORDER  BY EMP_ID;
  • "ขอคอลัมน์รหัส+ชื่อ+อีเมล เฉพาะแถวที่ ROLE_ID = 4 (บทบาทพนักงานกลุ่มเป้าหมาย) เรียงตามรหัส" — ผลบนจอ = คนกลุ่มเป้าหมาย 4 คนนี้เอง (ตัวเลขที่ใช้ทดสอบยืนยันทั้งหน้า PL/SQL):

📐 Constraint 4 ประเภท — จากตัวอย่างจริงใน DDL นี้

ใน 05_VERIFY คอลัมน์ CONSTRAINT_TYPE จะมีแค่ 4 ตัวอักษร — ทุกคนต้องแปลได้ทันที:

ตัวอักษรชนิดความหมาย / กันอะไรตัวอย่างจริงจาก Part I
PPrimary Keyแถวตัวตนของตาราง ไม่ซ้ำ + ไม่ NULL — ใช้อ้างอิงจากตารางอื่น (เป็นแบบคอลัมน์เดียวหรือ composite ก็ได้)PK_DEPT · PK_DOC · PK_EMP — Part I มี PK ครบทั้ง 11 ตาราง
RForeign Key (Referential)ค่าต้องมีจริงในตารางแม่ — กันอ้างอิงของที่ไม่มีFK_DOC_CREATOR (ผู้ร่างต้องเป็นพนักงานที่มีอยู่)
UUniqueห้ามซ้ำในตาราง (แต่อนุญาต NULL ได้ต่างจาก PK)UQ_EMP_EMAIL · UQ_SIGN_DOC (ลงนาม 1 ครั้ง) · UQ_RLOG_DOC_EMP (ยืนยัน 1 ครั้ง)
CCheck / Not Nullจำกัดค่าให้อยู่ในเงื่อนไขธุรกิจ + คอลัมน์จำเป็นต้องมีค่าCK_DOC_STATUS (8 สถานะ) · CK_DOC_SLA (> 0) · NN_DOC_TITLE

🧠 ตัวเลข Constraint ของ Part I (นับจาก DDL จริง — ใช้ชี้ผล 05_VERIFY ได้)

P = 11 (ทุกตารางมี 1) · R = 14 · U = 4 · C = 31 (CHECK ที่ตั้งชื่อ CK_ 11 + NOT NULL NN_ 20) — รวม ~60 แถว Constraint ในผล VERIFY เมื่อรันเสร็จ — จำแค่ "11 ตาราง ตารางละ PK" + "FK 14 ตัว" ก็ตอบได้ว่าระบบผูกกันสมบูรณ์

Q ทำไม READ_LOGS ถึงใช้ UNIQUE (DOC_ID, EMP_ID) แทนที่จะให้ DOC_ID เป็น UNIQUE เหมือน SIGNATURES?

เฉลย: READ_LOGS = หลายแถวต่อเอกสาร (คนละคน) จึงล็อกที่คู่ (DOC_ID, EMP_ID) กันแค่ "คนเดิมยืนยันซ้ำ" · SIGNATURES = แถวเดียวต่อเอกสาร จึง UNIQUE แค่ DOC_ID

Q ใน 05_VERIFY เห็นแถว CONSTRAINT_TYPE = 'R' — แปลว่าอะไร?

เฉลย: R = Referential = Foreign Key จำคู่กัน: P = Primary, R = Reference (FK), U = Unique, C = Check/NOT NULL
